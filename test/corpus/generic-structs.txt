pub enum Node { Null, Number(f64), Link(*Node) }
type IntBox = Box<i32>
fn read<T>(box: m.Box<T>, pair: Pair<i32, []u8>) Box<T> {
  x: Box<T>
  case node {
    Node.Null => { }
    Node.Number(v) => { return v }
    Node.Link(*p) => { p.* = nil }
  }
  return x
}
