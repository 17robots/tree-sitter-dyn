enum FsErr {
  NotFound(str),
  PermissionDenied,
}

enum ParseErr {
  Empty,
  BadFormat(i32),
}

type AppErr = FsErr | ParseErr

fn load(path: str) i32!(FsErr | ParseErr) {
  errdefer cleanup()
  if path.len == 0 {
    fail ParseErr.Empty
  }
  v := try parse(path)
  return v
}

fn check_void() !FsErr {
  fail FsErr.PermissionDenied
}

fn untyped() i32! {
  fail
}

fn main() {
  x := load("a") catch 0
  y := load("b") catch err {
    case err {
      FsErr.NotFound(p) => { return }
      FsErr.PermissionDenied, ParseErr.Empty => { return }
      ParseErr(pe) => { return }
    }
  }
  check_void() catch {}
}
