use col "std/collections", raw "std/io", "std/mem"

enum Shape {
  Empty,
  Circle(f32),
}

fn demo(xs: []const u8, s: *Shape) usize {
  doc := `line 1
line 2 with "quotes" and \slashes`
  c: Shape = .Circle(2.0)
  outer: for i in 0..xs.len {
    for j in 0..4 {
      if i == j { continue outer }
      if j == 3 { break outer }
    }
  }
  case s.* {
    .Empty => { return doc.len }
    .Circle(*r) => { r.* += 1.0 }
  }
  if c == .Empty { return 0 }
  return xs.len
}
